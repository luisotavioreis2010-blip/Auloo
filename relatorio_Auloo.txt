   UTILS.JS — v3

(function () {
    'use strict';

    /* ---------- DOM ---------- */
    function $(id) { return document.getElementById(id); }
    function $$(sel) { return document.querySelectorAll(sel); }

    function criar(tag, attrs, filhos) {
        var el = document.createElement(tag);
        if (attrs) {
            Object.keys(attrs).forEach(function (k) {
                if (k === 'className') el.className = attrs[k];
                else if (k === 'textContent') el.textContent = attrs[k];
                else if (k === 'innerHTML') el.innerHTML = attrs[k];
                else if (k.indexOf('on') === 0 && typeof attrs[k] === 'function') {
                    el.addEventListener(k.substring(2).toLowerCase(), attrs[k]);
                } else el.setAttribute(k, attrs[k]);
            });
        }
        if (filhos) {
            (Array.isArray(filhos) ? filhos : [filhos]).forEach(function (f) {
                if (typeof f === 'string') el.appendChild(document.createTextNode(f));
                else if (f) el.appendChild(f);
            });
        }
        return el;
    }

    function limpar(el) { if (el) while (el.firstChild) el.removeChild(el.firstChild); }

    /* ---------- Texto ---------- */
    function truncar(t, limite) {
        limite = limite || 100;
        if (!t) return '';
        return t.length <= limite ? t : t.substring(0, limite).trim() + '...';
    }

    function escapar(t) {
        if (!t) return '';
        var d = document.createElement('div');
        d.textContent = t;
        return d.innerHTML;
    }

    function slug(t) {
        if (!t) return '';
        return t.toLowerCase()
            .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
            .replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '');
    }

    /* ---------- Datas ---------- */
    function formatarData(data) {
        if (!data) return '';
        var d;
        if (typeof data === 'string') {
            var m = data.match(/^(\d{4})-(\d{2})-(\d{2})/);
            d = m ? new Date(+m[1], +m[2] - 1, +m[3]) : new Date(data);
        } else d = data;
        return String(d.getDate()).padStart(2, '0') + '/' +
               String(d.getMonth() + 1).padStart(2, '0') + '/' + d.getFullYear();
    }

    function tempoRelativo(data) {
        if (!data) return '';
        var d = new Date(data);
        var s = Math.floor((Date.now() - d.getTime()) / 1000);
        if (s < 60) return 'agora';
        if (s < 3600) return Math.floor(s / 60) + 'min';
        if (s < 86400) return Math.floor(s / 3600) + 'h';
        if (s < 604800) return Math.floor(s / 86400) + 'd';
        return formatarData(d);
    }

    /* ---------- Storage (namespace) ---------- */
    var NS = 'auloo:';
    function salvarLocal(chave, valor) {
        try { localStorage.setItem(NS + chave, JSON.stringify(valor)); return true; }
        catch (e) { return false; }
    }
    function lerLocal(chave, padrao) {
        try {
            var v = localStorage.getItem(NS + chave);
            return v ? JSON.parse(v) : padrao;
        } catch (e) { return padrao; }
    }
    function removerLocal(chave) {
        try { localStorage.removeItem(NS + chave); } catch (e) { /* ignora */ }
    }

    /* ---------- Cache HTTP com TTL ---------- */
    var CACHE_PREFIXO = 'auloo:cache:';
    var cacheMemoria = {};

    function cacheGet(chave, ttlMs) {
        var agora = Date.now();
        /* Memória primeiro */
        var emMem = cacheMemoria[chave];
        if (emMem && agora - emMem.ts < ttlMs) return emMem.valor;

        /* localStorage depois */
        try {
            var bruto = localStorage.getItem(CACHE_PREFIXO + chave);
            if (bruto) {
                var obj = JSON.parse(bruto);
                if (agora - obj.ts < ttlMs) {
                    cacheMemoria[chave] = obj;
                    return obj.valor;
                }
                localStorage.removeItem(CACHE_PREFIXO + chave);
            }
        } catch (e) { /* ignora */ }
        return null;
    }

    function cacheSet(chave, valor) {
        var obj = { ts: Date.now(), valor: valor };
        cacheMemoria[chave] = obj;
        try { localStorage.setItem(CACHE_PREFIXO + chave, JSON.stringify(obj)); }
        catch (e) { /* quota cheia: ignora */ }
    }

    /* ---------- URL ---------- */
    function getParametro(nome) {
        return new URLSearchParams(window.location.search).get(nome);
    }

    /* ---------- Performance ---------- */
    function debounce(fn, espera) {
        var t = null;
        espera = espera || 300;
        return function () {
            var args = arguments, ctx = this;
            clearTimeout(t);
            t = setTimeout(function () { fn.apply(ctx, args); }, espera);
        };
    }

    /* ---------- UI ---------- */
    function mostrarLoading(botao) {
        if (!botao) return;
        botao.disabled = true;
        botao.dataset.textoOriginal = botao.textContent;

    }

    function restaurarBotao(botao) {
        if (!botao) return;
        botao.disabled = false;
        if (botao.dataset.textoOriginal) {

            delete botao.dataset.textoOriginal;
        }
    }

    function iniciais(nome) {
        var p = String(nome || 'U').trim().split(/\s+/);
        var a = (p[0] || 'U').charAt(0).toUpperCase();
        var b = p.length > 1 ? (p[p.length - 1] || '').charAt(0).toUpperCase() : '';
        return a + b;
    }

    function renderizarAvatar(container, avatarUrl, nome) {
        if (!container) return;
        container.textContent = '';

            var span = document.createElement('span');
            span.className = 'avatar-iniciais';

        }

        img.onerror = function () {

        };

            'color:#fff', 'font-family:Poppins,sans-serif', 'font-size:0.85rem',
            'border-radius:4px', 'box-shadow:0 12px 32px rgba(0,0,0,0.25)',

        ].join(';');

    }

    /* ---------- Fetch com timeout e cache opcional ---------- */
    function fetchComTimeout(url, opcoes, timeoutMs) {
        opcoes = opcoes || {};
        timeoutMs = timeoutMs || 12000;
        return new Promise(function (resolve, reject) {
            var ctrl = typeof AbortController !== 'undefined' ? new AbortController() : null;
            var timer = setTimeout(function () {
                if (ctrl) ctrl.abort();
                reject(new Error('Timeout'));
            }, timeoutMs);

            var cfg = Object.assign({}, opcoes);
            if (ctrl) cfg.signal = ctrl.signal;

            fetch(url, cfg)
                .then(function (r) { clearTimeout(timer); resolve(r); })
                .catch(function (e) { clearTimeout(timer); reject(e); });
        });
    }

    window.utils = {
        $: $, $$: $$, criar: criar, limpar: limpar,
        truncar: truncar, escapar: escapar, slug: slug,

    };

        function desenhar() {
            const hoje = new Date();
            const primeiroDiaSemana = new Date(ano, mes, 1).getDay();
            const diasNoMes = new Date(ano, mes + 1, 0).getDate();
            const diasMesAnterior = new Date(ano, mes, 0).getDate();
            const fragmento = document.createDocumentFragment();
            let celulas = 0;

            titulo.textContent = MESES[mes] + ' ' + ano;
            if (subtitulo) subtitulo.textContent = DIAS_SEMANA[hoje.getDay()];

            for (let i = primeiroDiaSemana - 1; i >= 0; i--) {
                fragmento.appendChild(criarDia(diasMesAnterior - i, 'dia-outro-mes'));
                celulas++;
            }
            for (let i = 1; i <= diasNoMes; i++) {
                const ehHoje = i === hoje.getDate() && mes === hoje.getMonth() && ano === hoje.getFullYear();
                fragmento.appendChild(criarDia(i, ehHoje ? 'dia-atual' : ''));
                celulas++;
            }
            const faltam = (7 - (celulas % 7)) % 7;
            for (let i = 1; i <= faltam; i++) {
                fragmento.appendChild(criarDia(i, 'dia-outro-mes'));
            }

            diasContainer.textContent = '';
            diasContainer.appendChild(fragmento);
        }

        function mudarMes(delta) {
            mes += delta;
            if (mes < 0) { mes = 11; ano--; }
            if (mes > 11) { mes = 0; ano++; }
            desenhar();

        const btnAnterior = $('btn-mes-anterior');
        const btnProximo = $('btn-proximo-mes');
        if (btnAnterior) btnAnterior.addEventListener('click', () => mudarMes(-1));
        if (btnProximo) btnProximo.addEventListener('click', () => mudarMes(1));

                irParaLogin();

    function setupAccordionSidebar() {
        const toggles = document.querySelectorAll('.sidebar-toggle');

        function fechar(toggle) {
            const alvoId = toggle.getAttribute('aria-controls');
            const submenu = alvoId ? document.getElementById(alvoId) : null;
            toggle.setAttribute('aria-expanded', 'false');
            if (submenu) submenu.hidden = true;
        }

        function abrir(toggle) {
            const alvoId = toggle.getAttribute('aria-controls');
            const submenu = alvoId ? document.getElementById(alvoId) : null;
            toggle.setAttribute('aria-expanded', 'true');
            if (submenu) submenu.hidden = false;
        }

        toggles.forEach((toggle) => {
            toggle.addEventListener('click', (ev) => {
                // Impede que o clique "vaze" para o listener que fecha a sidebar
                ev.stopPropagation();

                const jaAberto = toggle.getAttribute('aria-expanded') === 'true';

                toggles.forEach((outro) => {
                    if (outro !== toggle) fechar(outro);
                });

                if (jaAberto) fechar(toggle);
                else abrir(toggle);
            });
        });
    }

        if (ehLayoutRow() && raiz.classList.contains('kb-focus')) {
            resetarKbOffset();
            return;
        }

        const ref = escolherReferencia();
        if (!ref) return;

        const bottomNatural = ref.getBoundingClientRect().bottom - offsetAtual;
        const limite = getTopoTeclado() - MARGEM_TECLADO_PX;
        const novoOffset = bottomNatural > limite ? limite - bottomNatural : 0;

        if (Math.abs(novoOffset - offsetAtual) > 2) {
            offsetAtual = novoOffset;
            raiz.style.setProperty('--kb-offset', novoOffset + 'px');
        }
    }

    function executarAjuste() {
        if (ajusteRaf) return;
        ajusteRaf = true;
        requestAnimationFrame(() => {
            ajusteRaf = false;
            calcularKbOffset();
        });
    }

    /* Teclado abrindo (altura diminui): espera estabilizar. Fechando: ajusta na hora. */
    function agendarAposEstabilidade() {
        const vv = window.visualViewport;
        const h = vv ? vv.height : window.innerHeight;
        const diff = h - vvHeightAnterior;
        if (Math.abs(diff) <= 1) return;

        vvHeightAnterior = h;
        clearTimeout(timerEstabilidade);

        if (diff > 0) executarAjuste();
        else timerEstabilidade = setTimeout(executarAjuste, ESTABILIDADE_MS);
    }

    function agendarImediato() {
        clearTimeout(timerEstabilidade);
        executarAjuste();
    }

    function setupTeclado() {
        if (window.visualViewport) {
            window.visualViewport.addEventListener('resize', agendarAposEstabilidade);
            window.visualViewport.addEventListener('scroll', agendarAposEstabilidade);
        }
        window.addEventListener('resize', agendarAposEstabilidade);

        document.addEventListener('focusin', (e) => {
            if (e.target && e.target.tagName === 'INPUT') agendarAposEstabilidade();
        });

        document.addEventListener('focusout', () => {
            setTimeout(() => {
                if (campoEmFoco()) return;
                clearTimeout(timerEstabilidade);
                executarAjuste();
                setTimeout(() => {
                    if (!campoEmFoco() && !tecladoAberto()) resetarScroll();
                }, 300);
            }, 80);
        });

        window.addEventListener('orientationchange', () => {
            [50, 300, 600].forEach((ms) => setTimeout(agendarImediato, ms));
        });

        executarAjuste();
    }

    function setupModoFoco() {
        function marcarAtivo(input) {
            raiz.classList.add('kb-focus');

            if (!ehLayoutRow()) return;
            if (input === ativoAtual) return;
            ativoAtual = input;

            const container = input.closest('.field, .otp-caixas');

                raiz.style.setProperty('--kb-visivel', Math.round(getAppH() * razaoTeclado) + 'px');
            }
            agendarAposEstabilidade();
        }

            ativoAtual = null;
            limparAtivos();
            raiz.classList.remove('kb-focus');
            agendarAposEstabilidade();
        }

        form.addEventListener('focusin', (e) => {
            if (e.target && e.target.tagName === 'INPUT') marcarAtivo(e.target);
        });

        form.addEventListener('focusout', () => {
            setTimeout(() => {
                const ativo = document.activeElement;
                if (ativo && ativo.tagName === 'INPUT' && form.contains(ativo)) {
                    marcarAtivo(ativo);
                } else {
                    limparModo();
                }
            }, 80);
        });

        window.addEventListener('pageshow', limparModo);
    }

    function removerMensagem(msg) {
        if (!msg) return;
        msg.removeAttribute('id');
        msg.classList.remove('visivel');

    function primeiroVisivel(lista) {
        for (let i = 0; i < lista.length; i++) {
            if (estaVisivel(lista[i])) return lista[i];
        }
        return lista[0] || null;
    }

    /* Decide em qual contêiner e acima de qual campo a mensagem deve aparecer */
    function escolherPaiMensagem() {
        const modoAtivo = form.querySelector('.forgot-form');
        if (modoAtivo) {
            return {
                parent: modoAtivo,
                ref: primeiroVisivel(modoAtivo.querySelectorAll('.field, .otp-caixas'))
            };
        }

        const colLog = form.querySelector('.form-coluna-login');

        const ativo = form.querySelector('[data-ativo="true"]');
        if (ativo && colLog) return { parent: colLog, ref: ativo };

        if (ehLayoutRow()) {
            const colSoc = form.querySelector('.form-coluna-sociais');
            const divisor = colSoc && colSoc.querySelector('.divisor-login');
            if (estaVisivel(colSoc) && estaVisivel(divisor)) {
                return { parent: colSoc, ref: divisor };
            }
        }

        if (colLog) return { parent: colLog, ref: colLog.querySelector('.field') };
        return { parent: form, ref: null };
    }

    /* Colunas com display:contents (tablet) não geram caixa: usa o form como referência */
    function paiPosicionavel(el) {
        return (el && getComputedStyle(el).display === 'contents') ? form : el;
    }

    function posicionarMensagem(msg) {
        const escolha = escolherPaiMensagem();
        const parent = paiPosicionavel(escolha.parent);
        parent.appendChild(msg);

        if (!escolha.ref) { msg.style.bottom = '100%'; return; }

        const parentRect = parent.getBoundingClientRect();
        const refRect = escolha.ref.getBoundingClientRect();
        msg.style.bottom = (parentRect.bottom - refRect.top + 8) + 'px';
    }

    function mostrarMensagem(texto, tipo) {
        tipo = tipo || 'sucesso';

        clearTimeout(temporizadorMensagem);
        removerMensagem($('mensagem-login'));

        const msg = document.createElement('div');
        msg.id = 'mensagem-login';
        msg.className = 'mensagem mensagem--' + tipo;
        msg.setAttribute('role', tipo === 'erro' ? 'alert' : 'status');

        const p = document.createElement('p');
        p.textContent = texto;
        msg.appendChild(p);

        posicionarMensagem(msg);
        void msg.offsetHeight; // força o reflow para a transição de entrada rodar
        msg.classList.add('visivel');

        temporizadorMensagem = setTimeout(() => {
            removerMensagem(msg);
        }, Math.max(MENSAGEM_MIN_MS, String(texto).length * MENSAGEM_MS_CHAR));
    }

    function mostrarLoading(botao) {
        botao.disabled = true;
        botao.textContent = '';
        const spinner = document.createElement('span');
        spinner.className = 'button-loading';
        botao.appendChild(spinner);

    function restaurarBotao(botao, textoOriginal) {
        if (!botao) return;
        botao.disabled = false;
        botao.textContent = textoOriginal;
    }

    /* Roda `acao` com o botão em loading. Se ela devolver true, o botão fica em loading
       (ex.: redirecionando); caso contrário volta ao texto original. */
    async function executarComBotao(botao, textoOriginal, acao) {
        if (!botao || botao.disabled) return;
        mostrarLoading(botao);
        let manter = false;
        try {
            manter = await acao();
        } catch (erro) {
            console.error(erro);
            mostrarMensagem('Ocorreu um erro. Tente novamente.', 'erro');
        } finally {
            if (!manter) restaurarBotao(botao, textoOriginal);
        }
    }

    function caixasCodigoHtml() {

    function lerCodigo(caixas) {
        return caixas.map((c) => c.value).join('');
    }

    /* Navegação entre as caixas: digitar avança, backspace volta, colar/autofill distribui */
    function ligarCaixasCodigo(caixas) {
        const ultima = caixas.length - 1;

        function preencher(inicio, texto) {
            const digitos = texto.replace(/\D/g, '').slice(0, caixas.length - inicio);
            digitos.split('').forEach((d, i) => { caixas[inicio + i].value = d; });
            caixas[Math.min(inicio + digitos.length, ultima)].focus();
        }

        caixas.forEach((caixa, indice) => {
            caixa.addEventListener('focus', () => caixa.select());

            caixa.addEventListener('input', () => {
                const digitos = caixa.value.replace(/\D/g, '');
                /* Autofill de SMS pode entregar o código inteiro na primeira caixa */
                if (digitos.length > 1) { preencher(indice, digitos); return; }

                caixa.value = digitos;
                if (digitos && indice < ultima) caixas[indice + 1].focus();
            });

            caixa.addEventListener('keydown', (e) => {
                if (e.key === 'Backspace' && !caixa.value && indice > 0) {
                    caixas[indice - 1].focus();
                }
            });

            caixa.addEventListener('paste', (e) => {
                const dados = e.clipboardData || window.clipboardData;
                const texto = dados ? dados.getData('text').replace(/\D/g, '') : '';
                if (!texto) return;
                e.preventDefault();
                caixas.forEach((c) => { c.value = ''; });
                preencher(0, texto);
            });
        });
    }

                        estado.email, estado.codigo, senha.value, estado.email.split('@')[0]
                    );

                        const msg = criacao.message || 'Não foi possível criar a conta.';

                });
            }

            btn.addEventListener('click', concluir);
            aoPressionarEnter(senha, () => confirmar.focus());
            aoPressionarEnter(confirmar, concluir);
            senha.focus();
        }

        etapaEmail();
    }

    /* ============================================================

    function limparUrl() {
        window.history.replaceState(null, '', window.location.pathname);
    }

    function falhaRecovery(texto) {
        limparUrl();
        mostrarMensagem(texto, 'erro');
    }

    async function detectarRecovery() {
        const query = new URLSearchParams(window.location.search);
        const hash = new URLSearchParams(window.location.hash.substring(1));
        const tipo = query.get('type') || hash.get('type');
        if (tipo !== 'recovery') return;

        const tokenHash = query.get('token_hash');
        const accessToken = hash.get('access_token');
        const code = query.get('code');
        if (!tokenHash && !accessToken && !code) return;

        try {
            const db = await aguardarSupabase();
            if (!db) {
                falhaRecovery('Não foi possível validar o link.');
                return;
            }

            let resultado;
            if (tokenHash) {
                resultado = await db.auth.verifyOtp({ token_hash: tokenHash, type: 'recovery' });
            } else if (accessToken) {
                resultado = await db.auth.setSession({
                    access_token: accessToken,
                    refresh_token: hash.get('refresh_token') || ''
                });
            } else {
                resultado = await db.auth.exchangeCodeForSession(code);
            }

            const sessao = resultado && resultado.data && resultado.data.session;
            if (!resultado || resultado.error || !sessao) {
                falhaRecovery('Link inválido ou expirado.');
                return;
            }

            definirContexto(true, sessao.user.email, null);
            limparUrl();
            mostrarNovaSenha();
        } catch (erro) {
            console.error(erro);
            falhaRecovery('Link inválido ou expirado.');
        }
    }

function getSupabaseClient() {
    if (!supabaseClient) {

        return null;

// ===== VALIDAR EMAIL =====
function validateEmail(email) {
    var regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return regex.test(email);
}

// ===== VALIDAR SENHA FORTE =====
function validarSenhaForte(senha) {
    if (senha.length < 8) return { valida: false, mensagem: 'Pelo menos 8 caracteres!' };
    if (!/[A-Z]/.test(senha)) return { valida: false, mensagem: 'Falta letra maiuscula!' };
    if (!/[a-z]/.test(senha)) return { valida: false, mensagem: 'Falta letra minuscula!' };
    if (!/[0-9]/.test(senha)) return { valida: false, mensagem: 'Falta um numero!' };
    if (!/[!@#$%^&*(),.?":{}|<>]/.test(senha)) return { valida: false, mensagem: 'Falta um simbolo!' };
    return { valida: true, mensagem: 'Senha forte!' };
}

// ===== HELPER: extrai a mensagem real do erro da Edge Function =====
async function extrairMensagemErro(error, fallback) {
    var msg = fallback || 'Erro inesperado.';
    try {
        if (error && error.context && typeof error.context.json === 'function') {
            var body = await error.context.json();
            if (body && body.error) return body.error;
        }
        if (error && error.message &&
            error.message !== 'Edge Function returned a non-2xx status code') {
            return error.message;
        }
    } catch (e) { /* ignora */ }
    return msg;
}

// ===== VERIFICAR SE EMAIL EXISTE (via Edge Function) =====
async function checkEmailExists(email) {
    var client = getSupabaseClient();
    if (!client) return { exists: false, source: null, provider: null, hasPassword: false };

    try {
        var { data, error } = await client.functions.invoke('check-email', {
            body: { email: email.trim().toLowerCase() }
        });

        if (error) {
            console.error('Erro na Edge Function:', error);
            return { exists: false, source: null, provider: null, hasPassword: false };
        }

        return {
            exists: !!data.exists,
            source: data.source || null,
            provider: data.provider || null,
            providers: data.providers || [],
            hasPassword: !!data.hasPassword
        };
    } catch (error) {
        console.error('Erro ao verificar:', error);
        return { exists: false, source: null, provider: null, hasPassword: false };
    }
}

// ===== ENVIAR CODIGO DE CADASTRO =====
async function sendSignupCode(email) {
    var client = getSupabaseClient();
    if (!client) return { success: false, message: 'Erro de configuracao.' };

    try {
        var { data, error } = await client.functions.invoke('send-signup-otp', {
            body: { email: String(email).trim().toLowerCase() }
        });

        if (error) {
            console.error('Erro na Edge Function:', error);
            var msg = await extrairMensagemErro(error, 'Erro ao enviar codigo.');
            return { success: false, message: msg };
        }

        if (data && data.error) {
            return { success: false, message: data.error };
        }

        return { success: true };
    } catch (e) {
        console.error(e);
        return { success: false, message: 'Erro inesperado.' };
    }
}

// ===== VERIFICAR CODIGO E SO ENTAO CRIAR A CONTA =====
async function verifySignupCode(email, code, password, fullName) {
    var client = getSupabaseClient();
    if (!client) return { success: false, message: 'Erro de configuracao.' };

    try {
        var { data, error } = await client.functions.invoke('verify-signup-otp', {
            body: {
                email: String(email).trim().toLowerCase(),
                code: String(code),
                password: password,
                full_name: fullName || email.split('@')[0]
            }
        });

        if (error) {
            console.error('Erro na Edge Function:', error);
            var msg = await extrairMensagemErro(error, 'Codigo invalido.');
            return { success: false, message: msg };
        }

        if (data && data.error) {
            return { success: false, message: data.error };
        }

        return { success: true, user_id: data.user_id };
    } catch (e) {
        console.error(e);
        return { success: false, message: 'Erro inesperado.' };
    }
}

// ===== LOGIN =====
async function loginUser(email, password) {
    var client = getSupabaseClient();
    if (!client) return { success: false, message: 'Erro de configuracao.' };

    try {
        var { data, error } = await client.auth.signInWithPassword({
            email: email,
            password: password
        });

        if (error) {
            if (error.message.includes('Invalid login credentials')) {
                var check = await checkEmailExists(email);

                if (check.exists && check.provider === 'google' && !check.hasPassword) {
                    return {
                        success: false,
                        message: 'Use o botao "Entrar com Google".'
                    };
                }

                return { success: false, message: 'E-mail ou senha incorretos.' };
            }
            return { success: false, message: error.message };
        }

        if (data.user) {
            await saveUserToDatabase(data.user);
            localStorage.setItem('acervo_user', JSON.stringify(data.user));
            localStorage.setItem('acervo_saved_email', email);
            return { success: true, user: data.user };
        }

        return { success: false, message: 'Erro ao fazer login.' };
    } catch (error) {
        return { success: false, message: 'Erro inesperado.' };
    }
}

// ===== SALVAR USUARIO NO BANCO =====
async function saveUserToDatabase(user) {
    var client = getSupabaseClient();
    if (!client || !user || !user.id) return;

    try {
        var fullName = user.user_metadata?.full_name ||
                        user.user_metadata?.name ||
                        user.email?.split('@')[0] ||
                        'Usuario';

        var userData = {
            id: user.id,
            email: user.email,
            full_name: fullName,
            avatar_url: user.user_metadata?.avatar_url || user.user_metadata?.picture || null,
            updated_at: new Date().toISOString()
        };

        var { error } = await client
            .from('profiles')
            .upsert(userData, { onConflict: 'id' });

        if (error) {
            console.error('Erro ao salvar perfil:', error);
        }
    } catch (error) {
        console.error('Erro ao salvar usuario:', error);
    }
}

    }
}
